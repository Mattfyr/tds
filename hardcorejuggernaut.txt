local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TeleportService = game:GetService("TeleportService")

local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")
local RemoteFunction = ReplicatedStorage:WaitForChild("RemoteFunction")

local LOBBY_PLACE_ID = 3260590327
local MAX_LOSSES = 30

local ActiveStratThread = nil
local StratRunning = false
local Teleported = false
local LossCount = 0

local function TeleportLobby()
    if Teleported then
        return
    end

    Teleported = true
    print("30 losses reached. Rejoining lobby...")

    local Success, Error = pcall(function()
        TeleportService:Teleport(LOBBY_PLACE_ID, LocalPlayer)
    end)

    if not Success then
        warn("Teleport failed:", Error)
        Teleported = false
    end
end

local function StartStrategy()
    if StratRunning or Teleported then
        return
    end

    StratRunning = true

    print("Starting strategy...")

    TDS:ResetIndex()
    TDS:Place("Pyromancer", -2.0989227294921875, 1.009635329246521, -34.57014465332031)
    TDS:SetTarget(1, "Farthest")
    TDS:Ready()

    TDS:VoteSkip(1, 25)
    TDS:Upgrade(1)
    TDS:Upgrade(1)

    TDS:VoteSkip(6)
    TDS:Upgrade(1)

    TDS:Upgrade(1)

    TDS:Place("Hunter", 1.5090651512145996, 0.5713815689086914, 14.419164657592773)
    TDS:Place("Hunter", 3.570600986480713, 0.9113662242889404, 18.20372200012207)
    TDS:VoteSkip(14)
    TDS:Place("Hunter", 0.24097543954849243, 1.317692518234253, 18.51652717590332)

    TDS:VoteSkip(15)
    TDS:Place("Hunter", 1.720466136932373, 1.384239912033081, 22.47343635559082)
    TDS:Place("Hunter", 5.2701520919799805, 1.3515396118164062, 22.529003143310547)
    TDS:Place("Hunter", 8.8175687789917, 1.3109925985336304, 22.54141616821289)
    TDS:Place("Hunter", 9.301279067993164, 0.847939133644104, 18.969970703125)
    TDS:Place("Hunter", 6.011168479919434, 1.3946723937988281, 28.527118682861328)
    TDS:Place("Hunter", 2.5719375610351562, 1.4263112545013428, 27.979778289794922)
    TDS:Place("Hunter", -1.284123420715332, 1.4632039070129395, 28.251195907592773)

    TDS:WaitForWave(23)

    for Index = 1, 11 do
        TDS:Sell(Index)
    end

    TDS:Place("EvolvedJuggernaut", -1.284123420715332, 1.4632039070129395, 28.251195907592773)

    StratRunning = false
    print("Strategy finished.")
end

local function GetMatchStatus()
    local Success, Status = pcall(function()
        local UiRoot = PlayerGui:FindFirstChild("ReactGameNewRewards")
        if not UiRoot then
            return nil
        end

        local MainFrame = UiRoot:FindFirstChild("Frame")
        if not MainFrame or not MainFrame.Visible then
            return nil
        end

        local GameOver = MainFrame:FindFirstChild("gameOver")
        if not GameOver or not GameOver.Visible then
            return nil
        end

        local RewardsScreen = GameOver:FindFirstChild("RewardsScreen")
        if not RewardsScreen or not RewardsScreen.Visible then
            return nil
        end

        local TopBanner = RewardsScreen:FindFirstChild("RewardBanner")
        local TextLabel = TopBanner and TopBanner:FindFirstChild("textLabel")

        if not TextLabel then
            return nil
        end

        local Text = TextLabel.Text:upper()
        print("Banner text:", Text)

        if Text:find("LOST") then
            return "LOSS"
        end

        return nil
    end)

    if Success then
        return Status
    end

    return nil
end

local function FireVotingSkipUntilGone()
    print("Firing Voting Skip until reward screen disappears...")

    while GetMatchStatus() ~= nil and not Teleported do
        local Success, Error = pcall(function()
            RemoteFunction:InvokeServer("Voting", "Skip")
        end)

        if Success then
            print("Voting Skip fired")
        else
            warn("Voting Skip failed:", Error)
        end

        task.wait(0.5)
    end

    print("Reward screen disappeared.")
end

task.spawn(function()
    ActiveStratThread = task.spawn(StartStrategy)

    while not Teleported do
        task.wait(0.5)

        local CurrentStatus = GetMatchStatus()

        if CurrentStatus == "LOSS" then
            LossCount += 1
            print("Loss Count:", LossCount .. "/" .. MAX_LOSSES)

            print("Loss detected! Stopping old thread...")

            if ActiveStratThread and coroutine.status(ActiveStratThread) ~= "dead" then
                task.cancel(ActiveStratThread)
                ActiveStratThread = nil
            end

            StratRunning = false

            if LossCount >= MAX_LOSSES then
                TeleportLobby()
                break
            end

            FireVotingSkipUntilGone()

            task.wait(1.5)

            print("Starting new strategy match!")
            ActiveStratThread = task.spawn(StartStrategy)
        end
    end
end)
