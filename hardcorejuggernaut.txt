local StratRunning = false

    RemoteFunction:InvokeServer("Multiplayer", "v2:start", {
        difficulty = "Easy",
        mode = "hardcore",
        count = 1,
    })

        return

    StratRunning = true

    task.wait(2.5)
    TDS:RemoveIndex()
    TDS:Place("Pyromancer", -2.0989227294921875, 1.009635329246521, -34.57014465332031)

    TDS:VoteSkip(6)

    TDS:Place("Hunter", 1.5090651512145996, 0.5713815689086914, 14.419164657592773)

    TDS:Place("EvolvedJuggernaut", -1.284123420715332, 1.4632039070129395, 28.251195907592773)

    StratRunning = false
end

local function GetMatchStatus()
    local Success, Status = pcall(function()
        local UiRoot = PlayerGui:FindFirstChild("ReactGameNewRewards")
        if not UiRoot then
            return nil
        end

        local MainFrame = UiRoot:FindFirstChild("Frame")
        if not MainFrame or not MainFrame.Visible then
            return nil
        end

        local GameOver = MainFrame:FindFirstChild("gameOver")
        if not GameOver or not GameOver.Visible then
            return nil
        end

        local RewardsScreen = GameOver:FindFirstChild("RewardsScreen")
        if not RewardsScreen or not RewardsScreen.Visible then
            return nil
        end

        local TopBanner = RewardsScreen:FindFirstChild("RewardBanner")
        if not TopBanner then
            return nil
        end

        local Label = TopBanner:FindFirstChild("textLabel") or TopBanner:FindFirstChildOfClass("TextLabel")
        if not Label then
            return nil
        end

        local Text = Label.Text:upper()
        if Text == "" then
            return nil
        end

        if Text:find("TRIUMPH", 1, true) or Text:find("VICTORY", 1, true) or Text:find("WIN", 1, true) then
            return "WIN"
        end

        if Text:find("LOST", 1, true) or Text:find("DEFEAT", 1, true) or Text:find("FAIL", 1, true) then
            return "LOSS"
        end

        return nil
    end)

    if Success then
        return Status
    end

    return nil
end

local function FireVotingSkipUntilGone()
    while GetMatchStatus() ~= nil and not Teleported do
        pcall(function()
            RemoteFunction:InvokeServer("Voting", "Skip")
        end)

        task.wait(0.5)
    end
end

task.spawn(function()
    ActiveStratThread = task.spawn(StartStrategy)

    while not Teleported do
        task.wait(0.5)

        local CurrentStatus = GetMatchStatus()
        if CurrentStatus == "LOSS" then
            LossCount += 1

            if ActiveStratThread and coroutine.status(ActiveStratThread) ~= "dead" then
                task.cancel(ActiveStratThread)
                ActiveStratThread = nil
            end

            StratRunning = false

            if LossCount >= MaxLosses then
                Teleported = true
                pcall(function()
                    TeleportService:Teleport(LobbyPlaceId, LocalPlayer)
                end)
                break
            end

            FireVotingSkipUntilGone()
            task.wait(1.5)

            ActiveStratThread = task.spawn(StartStrategy)
        end
    end
end)
