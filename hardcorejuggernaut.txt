local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TeleportService = game:GetService("TeleportService")

local LocalPlayer = Players.LocalPlayer
local GameState = require(ReplicatedStorage.Shared.Modules.GameState)
local Network = require(ReplicatedStorage.Shared.Modules.Network)
local Maid = require(ReplicatedStorage.Shared.Modules.Maid)

local LOBBY_PLACE_ID = 3260590327
local MAX_LOSSES = 30

local activeStratThread = nil
local stratRunning = false
local teleported = false
local lossCount = 0
local handlingMatchEnd = false
local restartPending = false
local MatchMaid = Maid.new()

local function GetLocalTeam()
    return LocalPlayer.Team
end

local function GetMatchOutcome()
    local Replicator = GameState.Replicator
    if Replicator == nil then
        return nil
    end

    if Replicator:Get("GameOver") ~= true then
        return nil
    end

    local WinningTeam = Replicator:Get("WinningTeam")
    if WinningTeam == nil then
        return nil
    end

    local LocalTeam = GetLocalTeam()
    if WinningTeam == LocalTeam or tostring(WinningTeam) == tostring(LocalTeam) then
        return "WIN"
    end

    return "LOSS"
end

local function TeleportLobby()
    if teleported then
        return
    end

    teleported = true
    MatchMaid:Sweep()
    print("30 losses reached. Rejoining lobby...")

    local success, err = pcall(function()
        TeleportService:Teleport(LOBBY_PLACE_ID, LocalPlayer)
    end)

    if not success then
        warn("Teleport failed:", err)
        teleported = false
    end
end

local function FireVotingSkip()
    pcall(function()
        Network.Channel("Voting"):InvokeServer("Skip")
    end)
end

local function StartStrategy()
    if stratRunning or teleported then
        return
    end

    stratRunning = true

    print("Starting strategy...")

    TDS:ResetIndex()
    TDS:Ready()

    TDS:Place("Pyromancer", -5.936307430267334, 0.9551397562026978, -31.831748962402344)
    TDS:SetTarget(1, "Farthest")

    for i = 1, 4 do
        TDS:Upgrade(1)
    end

    for i = 1, 10 do
        TDS:Place("Hunter", 2.679039716720581, 1.341203212738037, 21.30302619934082, true)
    end

    TDS:Place("EvolvedJuggernaut", 2.679039716720581, 1.341203212738037, 21.30302619934082, true)

    stratRunning = false
    print("Strategy finished.")
end

local function HandleMatchEnd()
    if handlingMatchEnd or teleported then
        return
    end

    local Outcome = GetMatchOutcome()
    if Outcome ~= "LOSS" then
        return
    end

    handlingMatchEnd = true
    restartPending = true
    lossCount += 1

    if activeStratThread and coroutine.status(activeStratThread) ~= "dead" then
        task.cancel(activeStratThread)
        activeStratThread = nil
    end

    stratRunning = false

    if lossCount >= MAX_LOSSES then
        TeleportLobby()
        handlingMatchEnd = false
        return
    end

    handlingMatchEnd = false
    FireVotingSkip()
end

local function StartNewStrategyIfNeeded()
    if teleported or not restartPending then
        return
    end

    restartPending = false
    task.wait(1.5)
    activeStratThread = task.spawn(StartStrategy)
end

local function BindGameStateSignals(Replicator)
    MatchMaid:Sweep()

    MatchMaid:Mark(Replicator:GetStateChangedSignal("GameOver"):Connect(function()
        task.spawn(HandleMatchEnd)
    end))

    MatchMaid:Mark(Replicator:GetStateChangedSignal("WinningTeam"):Connect(function()
        task.spawn(HandleMatchEnd)
    end))

    MatchMaid:Mark(Replicator:GetStateChangedSignal("GameStarted"):Connect(function(GameStarted)
        if GameStarted == true then
            task.spawn(StartNewStrategyIfNeeded)
        end
    end))

    if Replicator:Get("GameOver") == true then
        task.spawn(HandleMatchEnd)
    end
end

GameState:GetState():andThen(BindGameStateSignals)

task.spawn(function()
    activeStratThread = task.spawn(StartStrategy)
end)
